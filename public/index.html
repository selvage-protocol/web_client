// A guest opens the document the room shares.
fn main() {
    let invite = Invite::from_env();
    let mut room = Room::connect(invite);

    room.on_open(|path: &str| {
        println!("opened {path}");
    });

    for peer in room.peers() {
        let caret = peer.caret();
        println!("{peer} is on line {}", caret.line);
    }

    let notes = room.open("notes.md");
    notes.insert(0, "hello from the room\n");

    room.run();
}

// The room ends when the host leaves; a guest waits out the grace.
fn grace() -> Duration {
    Duration::from_secs(30)
}

/// The paths the host granted, in the room's own order.
fn granted(room: &Room) -> Listing {
    room.listing()
        .iter()
        .map(|entry| entry.path())
        .filter(|path| path.shared())
        .collect()
}
